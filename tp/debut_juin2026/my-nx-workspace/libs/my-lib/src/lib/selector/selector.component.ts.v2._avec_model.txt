import { Component, input, model, output, SimpleChange } from '@angular/core';

@Component({
  selector: 'lib-selector',
  imports: [],
  templateUrl: './selector.component.html',
  styleUrl: './selector.component.css',
})
export class SelectorComponent {
    titre=input("selection de ...");
    choixPossibles=input<string[]>([]);
    selectionInterne:string|null=null;
    
    choix=model<string>("");

    ngOnChanges(changes:SimpleChange){
      this.selectionInterne=this.choix();
    }

    onSelection(c:string){
       this.selectionInterne=c;
       this.choix.set(this.selectionInterne);
    }
}
